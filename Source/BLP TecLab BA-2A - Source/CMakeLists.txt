cmake_minimum_required(VERSION 3.22)
project(BLPTecLab VERSION 1.0.1)
set(CMAKE_CXX_STANDARD 17)
set(JUCE_DIR "" CACHE PATH "JUCE source directory")
add_subdirectory("${JUCE_DIR}" JUCE)
juce_add_plugin(BLPTecLab COMPANY_NAME BLP PLUGIN_MANUFACTURER_CODE Blpa PLUGIN_CODE Bt2a FORMATS VST3 Standalone PRODUCT_NAME "BLP TecLab BA-2A" VST3_CATEGORIES Fx Dynamics)
juce_generate_juce_header(BLPTecLab)
target_sources(BLPTecLab PRIVATE Plugin.cpp)
target_compile_definitions(BLPTecLab PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(BLPTecLab PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)

juce_add_console_app(BLPTests PRODUCT_NAME BLPTests)
juce_generate_juce_header(BLPTests)
target_sources(BLPTests PRIVATE Tests.cpp)
target_compile_definitions(BLPTests PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(BLPTests PRIVATE juce::juce_audio_utils)

option(BA2A_BUILD_HOST_TESTS "Build the actual VST3 render and regression test host" OFF)
if(BA2A_BUILD_HOST_TESTS)
 juce_add_console_app(BA2AHostTests PRODUCT_NAME "BA-2A Host Tests")
 juce_generate_juce_header(BA2AHostTests)
 target_sources(BA2AHostTests PRIVATE Validation/HostTests.cpp)
 target_compile_definitions(BA2AHostTests PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_MODAL_LOOPS_PERMITTED=1)
 target_link_libraries(BA2AHostTests PRIVATE juce::juce_audio_utils)
endif()
